//probeReadable.mjs — 呼叫端給的值之可讀性探針(內部模組, 不列入README之Functions; 帳本R-PROBE-01, 單一來源)
//
//【為何需要】本套件原借wsemi判斷函數(isestr、isbol、iseobj、isearr)對「讀取即拋錯之值」(已撤銷之Proxy、Symbol.toStringTag或陷阱拋錯者、元素存取器拋錯之陣列等)拋錯,
//  使寫入入口、selectByPk與建構層之防線成立(帳本R-WRITE-ERR-01: 任何因輸入導致之失敗皆走error事件恰一次後reject)。
//  wsemi 1.9.69起判斷函數對任何輸入皆回布林不拋錯(拋錯者回false, dspec〈輸入參數之檢測機制〉C2), 於是「不可讀」與「不是物件／字串／布林」於判斷函數上不可分,
//  不可讀之值被當成輸入無效而早退(回n:0或[], 不發事件)或被當成型別不符而忽略。
//
//【用法】只在判斷函數回false(無效或型別不符)時才探針, 用以分辨「不可讀」與「可讀而型別不符」: 不可讀者拋原錯誤、走該站點之原失敗通道, 可讀者回傳後照原行為(早退、忽略、預設)。
//  判斷函數回true(有效)者不探針: 不增加有效輸入之讀取次數與成本, 其後之cloneDeep等逐一讀取, 拋錯者同走錯誤路徑。
//
//【兩種探針】皆「不可讀者拋出原錯誤(訊息原樣, R-WRITE-ERR-01規則2)、可讀者回傳型別標籤」, 只讀型別標籤、鍵與描述子, 不執行值之任何方法、不呼叫存取器、不展開二進位內容:
//  probeTag: 型別標籤(Object.prototype.toString.call, 同修前wsemi isstr/isestr之讀法); 供純量判定站點(主鍵、option值、建構設定值、id值)
//  probeReadable: 型別標籤＋容器結構(同修前wsemi iseobj/isearr所讀者); 供「判定是否為物件／陣列」之站點(data)
//     物件(標籤為[object Object]): 沿原型鏈對每一層列自有鍵(ownKeys陷阱)、讀各字串鍵之描述子(getOwnPropertyDescriptor陷阱, 不呼叫存取器)、取原型(getPrototypeOf陷阱);
//       同修前iseobj之for...in所觸發者(for...in不處理Symbol鍵, 故只讀字串鍵之描述子)
//     陣列(標籤為[object Array]): length, 且length為1時讀元素[0](wsemi isearr僅此一元素之判定依賴其值; 其餘元素由其後之cloneDeep逐一讀取, 拋錯者同走錯誤路徑)
//     其餘標籤(Date、Map、Buffer、純量等)只讀標籤: 與修前判斷函數一致(其標籤已使判定為假), 不為此列舉大型二進位之鍵
//  已知邊界: 修前isbol(lodash isBoolean)另會觸發has陷阱(Symbol.toStringTag in v), 僅has陷阱拋錯之Proxy作為option.returnList、opt.useCache、opt.autoGenPk時, 修前拋錯, 現視為可讀而型別不符(同數字等, 走忽略通道)


/**
 * 讀取型別標籤: 已撤銷之Proxy、Symbol.toStringTag之存取器或get陷阱拋錯者拋出原錯誤
 *
 * @param {*} v 輸入呼叫端給予之值(純量恆可讀)
 * @returns {String} 回傳內部標籤字串，如'[object Object]'、'[object Array]'
 */
function probeTag(v) {
    return Object.prototype.toString.call(v)
}


/**
 * 可讀性探針: 判定呼叫端給的data等容器值可否被讀取(標籤＋容器結構), 不可讀者拋出原錯誤
 *
 * @param {*} v 輸入呼叫端給予之值(純量恆可讀)
 * @returns {String} 回傳內部標籤字串；不可讀時拋出原錯誤(訊息為拋出者之原訊息)，不回傳
 */
function probeReadable(v) {
    let tag = probeTag(v)
    if (tag === '[object Array]') {

        //length與元素[0], 同修前isearr之讀取; Reflect.get為明確之讀取(拋錯者拋)
        let n = Reflect.get(v, 'length')
        if (n === 1) {
            Reflect.get(v, 0)
        }

    }
    else if (tag === '[object Object]') {

        //沿原型鏈每一層之自有鍵、字串鍵之描述子(不呼叫存取器)與原型, 同修前iseobj之for...in所觸發者
        for (let p = v; p !== null; p = Reflect.getPrototypeOf(p)) {
            for (let k of Reflect.ownKeys(p)) {
                if (typeof k === 'string') {
                    Reflect.getOwnPropertyDescriptor(p, k)
                }
            }
        }

    }
    return tag
}


export { probeTag, probeReadable }
